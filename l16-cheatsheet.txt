 1. INTERNE ZONE IN BETRIEB NEHMEN
=====================================================================

--- Grundkonfiguration: Rekursion, Forwarder, Berechtigungen ---------

Auf SRV-DNS:
cd /etc/bind
ls                                   # named.conf = lädt .options und .local
sudo nano named.conf.options         # gesamten Inhalt durch den Block unten ersetzen

// Interne Netze: nur diese erhalten Antworten und Rekursion (Z5)
acl intern {
    10.10.20.0/24;
    <netz>/<präfix>;                 // weiteres internes Netz aus Z5
    <netz>/<präfix>;                 // weiteres internes Netz aus Z5
    localhost;
};

options {
    directory "/var/cache/bind";

    listen-on { 127.0.0.1; 10.10.30.53; };   // nur auf diesen Adressen
    listen-on-v6 { none; };                  // im Lab kein IPv6

    allow-query     { intern; };             // wer überhaupt fragen darf
    allow-recursion { intern; };             // wer externe Namen auflösen lassen darf (Z5)

    forwarders { 9.9.9.9; 149.112.112.112; }; // Quad9, zwei Forwarder (Z4)
    forward only;                            // NUR Forwarder, nie selbst ab der Root (Z4)

    dnssec-validation auto;                  // DNSSEC-Signaturen prüfen (Z5)
};


--- Zonen anmelden ----------------------------------------------------

--- Vorwärtszone techtrade.lab ----------------------------------------

Auf SRV-DNS:
sudo nano /etc/bind/zones/db.techtrade.lab

$TTL 3600
@   IN SOA ns1.techtrade.lab. hostmaster.techtrade.lab. (
        2026110101  ; Serial  JJJJMMTTnn - bei JEDER Änderung erhöhen
        7200        ; Refresh
        900         ; Retry
        1209600     ; Expire
        300 )       ; negative TTL (wie lange NXDOMAIN gecacht wird)
@        IN NS    ns1.techtrade.lab.
ns1      IN A     10.10.30.53
srv-dns  IN A     10.10.30.53
srv-web  IN A     10.10.20.13
srv-mon  IN A     10.10.100.20
r1       IN A     10.10.100.1
shop     IN CNAME srv-web            ; Alias (Z3) - ohne Punkt = relativ zur Zone

# Namen MIT Punkt am Ende sind absolut, Namen OHNE Punkt werden mit der Zone ergänzt.
# In Zonendateien beginnen Kommentare mit ; (nicht mit #).


--- Reverse-Zonen -----------------------------------------------------

# Eine Reverse-Zone pro Subnetz: IP-Adresse -> Name (PTR).
# Die Datei steht bereits für das Subnetz, in der Zeile steht nur das letzte Oktett.
# Der Name rechts endet IMMER mit einem Punkt, sonst hängt BIND die Zone an.

# Beispiel komplett: Server-LAN 10.10.20.0/24
Auf SRV-DNS:
sudo nano /etc/bind/zones/db.10.10.20

$TTL 3600
@   IN SOA ns1.techtrade.lab. hostmaster.techtrade.lab. ( 2026110101 7200 900 1209600 300 )
@   IN NS  ns1.techtrade.lab.
13  IN PTR srv-web.techtrade.lab.          ; 10.10.20.13 -> srv-web

# DNS-LAN 10.10.30.0/24 - Kopf bleibt gleich, Platzhalter mit Werten aus der Adressliste ersetzen
Auf SRV-DNS:
sudo nano /etc/bind/zones/db.10.10.30

$TTL 3600
@   IN SOA ns1.techtrade.lab. hostmaster.techtrade.lab. ( 2026110101 7200 900 1209600 300 )
@   IN NS  ns1.techtrade.lab.
<LETZTES_OKTETT>  IN PTR  <HOSTNAME>.techtrade.lab.

# Management-LAN 10.10.100.0/24 - zwei Systeme, also zwei PTR-Zeilen
Auf SRV-DNS:
sudo nano /etc/bind/zones/db.10.10.100

$TTL 3600
@   IN SOA ns1.techtrade.lab. hostmaster.techtrade.lab. ( 2026110101 7200 900 1209600 300 )
@   IN NS  ns1.techtrade.lab.
<LETZTES_OKTETT>  IN PTR  <HOSTNAME>.techtrade.lab.
<LETZTES_OKTETT>  IN PTR  <HOSTNAME>.techtrade.lab.


--- Prüfen und laden --------------------------------------------------

Auf SRV-DNS:

sudo named-checkzone techtrade.lab db.techtrade.lab           # «OK» am Ende
sudo named-checkzone 20.10.10.in-addr.arpa db.10.10.20

# named-checkzone <Zone> <Datei>: lädt die Datei wie BIND (Syntax, SOA, NS, CNAME-Konflikte).

# Inhaltlich falsche Einträge (falsche IP, fehlender Punkt) meldet die Prüfung NICHT.

sudo journalctl -u named -n 20 --no-pager # letzte 20 Logzeilen: «zone ... loaded serial ...»
# --no-pager = Ausgabe direkt ins Terminal (sonst Blätteransicht, Ende mit q)


--- Testen direkt am Server -------------------------------------------

Auf SRV-DNS:
dig @127.0.0.1 srv-web.techtrade.lab      # @ = diesen Server fragen; flags: qr aa rd ra
dig @127.0.0.1 shop.techtrade.lab         # CNAME + A in der Antwort

dig @127.0.0.1 -x 10.10.100.1 +short      # erwartet: r1.techtrade.lab.
dig @127.0.0.1 -x 10.10.100.20 +short     # erwartet: srv-mon.techtrade.lab.
# Steht am Ende «...techtrade.lab.30.10.10.in-addr.arpa.», fehlt der Punkt nach dem Namen.

# Wireshark öffnen (Startmenü -> Wireshark).
# Capture-Filter (Feld «...using this filter» vor dem Start):  port 53
# Schnittstelle ens3 doppelklicken -> Capture läuft.

Auf SRV-WEB:

dig @10.10.30.53 srv-web.techtrade.lab AAAA          # M5 (Typ AAAA)

--- Anzeigefilter in Wireshark ----------------------------------------

dns                                   # alle DNS-Pakete (auch SRV-DNS <-> Quad9)
dns && ip.addr == 10.10.20.13         # nur der Verkehr mit SRV-WEB, beide Richtungen
dns.flags.response == 0               # nur Anfragen
dns.flags.response == 1               # nur Antworten
ip.addr == 9.9.9.9 || ip.addr == 149.112.112.112      # Verkehr mit den Forwardern
dns.qry.name contains "example"       # Anfragen/Antworten zu einem Namen
dns.flags.rcode == 3                  # NXDOMAIN, in Wireshark «No such name»
                                      # (0 NOERROR, 2 SERVFAIL, 3 NXDOMAIN, 5 REFUSED)
dns.flags.response == 0 && not dns.response_in        # Anfragen ohne Antwort

# Paketdetails: «Domain Name System» aufklappen -> Transaction ID, Flags, Queries,
# Answers (TTL), Authoritative nameservers, [Request In: ...] / [Response In: ...]
# Antwortzeit: in der ANTWORT unten «[Time: 0.041 seconds]» - Einheit Sekunden (0.041 s = 41 ms)
# Als Spalte: Rechtsklick auf die Zeile [Time: ...] -> «Als Spalte anwenden»
# Die Spalte «Time» der Paketliste ist die Zeit seit Capture-Start, ebenfalls in Sekunden.
# Cache-Fehltreffer erkennen: zwischen Anfrage und Antwort liegen Pakete SRV-DNS <-> Quad9
# (Filter «dns» statt «ip.addr == 10.10.20.13» verwenden, sonst sind sie ausgeblendet).

--- Schnellüberblick: Statistiken -> DNS ------------------------------
# Wireshark -> Statistiken -> DNS
# rcode            = Verteilung der Antworten (No error, No such name, Server failure)
# Query/Response   = gleich viele Anfragen wie Antworten? Sonst gingen Antworten verloren
# request-response time (msec) = Antwortzeit Mittelwert / Min / Max
# Achtung: Ein Capture auf SRV-DNS enthält zwei Strecken (Client <-> SRV-DNS und
# SRV-DNS <-> Quad9). Nur Client-Sicht: unten im Feld «Anzeigefilter»
#   ip.addr == 10.10.20.13   eingeben -> «Anwenden».
# Es gibt nur den Mittelwert, kein p95 - dafür den tshark-Befehl in Auftrag 3 verwenden.

--- Auswertung: Grafana ----------------------------------------------

Auf SRV-MON (Firefox), Grafana -> Dashboards -> DNS-Dienst:
# Zeitraum oben rechts auf das Lastfenster setzen: «Absolute time range»,
#   From = Startzeit aus date, To = 5 Minuten später -> «Apply time range».
# «Anfragen gesamt (Zeitraum)»      = Anfragen im gewählten Zeitraum
# «Antworten je RCODE (Zeitraum)»   = NOERROR / NXDOMAIN / SERVFAIL im gewählten Zeitraum
#   Beide Panels rechnen mit increase(): ein Zähler-Reset (Neustart) wird erkannt.
# «Rohzähler»                       = Sprung nach unten beim Neustart von BIND
# «Exporter erreichbar (up)»        = 0 während der bind_exporter gestoppt war
# Plausibel? 300 s x 50 Anfragen/s = rund 15 000 Anfragen.

--- Auswertung: tshark ------------------------------------------------

Auf SRV-WEB:
cd ~
tshark -r last.pcap -Y 'dns.flags.response == 0' | wc -l
# Anfragen gesamt. -r Datei lesen, -Y Anzeigefilter, wc -l = Zeilen zählen

tshark -r last.pcap -Y 'dns.flags.response == 1' -T fields -e dns.flags.rcode | sort | uniq -c
# Antworten je RCODE: 0 NOERROR, 2 SERVFAIL, 3 NXDOMAIN
# -T fields -e Feld = nur dieses Feld ausgeben; sort | uniq -c = gleiche Werte zählen

tshark -2 -r last.pcap -Y 'dns.flags.response == 0 && not dns.response_in' | wc -l
# Anfragen ohne Antwort (Timeouts). -2 = zwei Durchgänge: erst im zweiten Durchgang
# weiss tshark bei einer Anfrage, ob später eine Antwort kam.
# Filter immer in einfachen Anführungszeichen ' ' - in " " würde bash ein ! umdeuten.

tshark -r last.pcap -Y 'dns.flags.response == 1' -T fields -e dns.time | sort -n > zeiten.txt
# alle Antwortzeiten in Sekunden, aufsteigend sortiert, in eine Datei (darum keine Ausgabe)
wc -l zeiten.txt                     # Anzahl Antwortzeiten in der Datei (rund 15 000)

LC_ALL=C awk '{s+=$1; a[NR]=$1} END {printf "Anzahl %d  Mittelwert %.1f ms  p50 %.1f ms  p95 %.1f ms  p99 %.1f ms\n", NR, s/NR*1000, a[int(NR*0.50)]*1000, a[int(NR*0.95)]*1000, a[int(NR*0.99)]*1000}' zeiten.txt
# awk liest zeiten.txt: Summe für den Mittelwert, Perzentil = Wert an Position NR*0.95 der sortierten Liste
# LC_ALL=C = Punkt als Dezimaltrennzeichen. Ohne diesen Zusatz liest awk auf der deutsch
# eingestellten VM «0.000312» als 0 und alle Werte werden 0,0.

--- Formel ------------------------------------------------------------
# Störungsquote = (SERVFAIL + Anfragen ohne Antwort) ÷ Anfragen gesamt
