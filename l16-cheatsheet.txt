 1. INTERNE ZONE IN BETRIEB NEHMEN
=====================================================================

--- Grundkonfiguration: Rekursion, Forwarder, Berechtigungen ---------

Auf SRV-DNS:
cd /etc/bind
ls                                   # named.conf = lädt .options und .local
sudo nano named.conf.options         # gesamten Inhalt durch den Block unten ersetzen

// Interne Netze: nur diese erhalten Antworten und Rekursion (Z5)
acl intern {
    10.10.20.0/24;
    10.10.30.0/24;
    10.10.100.0/24;
    localhost;
};

options {
    directory "/var/cache/bind";

    listen-on { 127.0.0.1; 10.10.30.53; };   // nur auf diesen Adressen
    listen-on-v6 { none; };                  // im Lab kein IPv6

    allow-query     { intern; };             // wer überhaupt fragen darf
    allow-recursion { intern; };             // wer externe Namen auflösen lassen darf (Z5)

    forwarders { 9.9.9.9; 149.112.112.112; }; // Quad9, zwei Forwarder (Z4)
    forward only;                            // NUR Forwarder, nie selbst ab der Root (Z4)

    dnssec-validation auto;                  // DNSSEC-Signaturen prüfen (Z5)
};

--- Auswertung: Grafana / Prometheus ----------------------------------

# Grafana: Dashboard DNS-Dienst, Zeitraum oben rechts auf die Messzeit setzen.
# Grafana -> Explore -> Datenquelle prometheus, Abfragen. Zeitraum = Messzeit,
# «Instant» einschalten und den Endzeitpunkt auf das Ende der Last setzen:

sum by (rcode) (increase(bind_response_rcodes_total{rcode=~"NOERROR|NXDOMAIN|SERVFAIL"}[5m]))
# Antworten je RCODE in den letzten 5 Minuten.
# increase = Zuwachs im Fenster; erkennt einen Zähler-Reset (Neustart) und rechnet weiter

sum by (result) (increase(bind_responses_total{result=~"Success|Nxrrset"}[5m]))
# Success = Antwort mit Eintrag, Nxrrset = NODATA (für die Erfolgsquote)

sum(increase(bind_incoming_queries_total[5m]))
# Anfragen gesamt im Fenster

sum by (rcode) (bind_response_rcodes_total{rcode=~"NOERROR|NXDOMAIN|SERVFAIL"})
# Rohzähler: Sprung nach unten beim Neustart von BIND

up{job="bind"}
# 0 während der bind_exporter gestoppt war

--- Auswertung: tshark ------------------------------------------------

Auf SRV-WEB:
cd ~
tshark -r last.pcap -Y "dns.flags.response == 0" | wc -l
# Anfragen gesamt. -r Datei lesen, -Y Anzeigefilter, wc -l = Zeilen zählen

tshark -r last.pcap -Y "dns.flags.response == 1" -T fields -e dns.flags.rcode | sort | uniq -c
# Antworten je RCODE: 0 NOERROR, 2 SERVFAIL, 3 NXDOMAIN
# -T fields -e Feld = nur dieses Feld ausgeben; sort | uniq -c = gleiche Werte zählen

tshark -r last.pcap -Y "dns.flags.response == 1 && dns.flags.rcode == 0 && dns.count.answers == 0" | wc -l
# NODATA: NOERROR ohne Eintrag

tshark -2 -r last.pcap -Y "dns.flags.response == 0 && !dns.response_in" | wc -l
# Anfragen ohne Antwort (Timeouts). -2 = zwei Durchgänge: erst im zweiten Durchgang
# weiss tshark bei einer Anfrage, ob später eine Antwort kam.

tshark -r last.pcap -Y "dns.flags.response == 1" -T fields -e dns.time | sort -n > zeiten.txt
# alle Antwortzeiten in Sekunden, aufsteigend sortiert, in eine Datei

awk '{s+=$1; a[NR]=$1} END {printf "Anzahl %d  Mittelwert %.1f ms  p50 %.1f ms  p95 %.1f ms  p99 %.1f ms\n", NR, s/NR*1000, a[int(NR*0.50)]*1000, a[int(NR*0.95)]*1000, a[int(NR*0.99)]*1000}' zeiten.txt
# awk liest zeiten.txt: Summe für den Mittelwert, Perzentil = Wert an Position NR*0.95 der sortierten Liste

--- Formeln -----------------------------------------------------------
# Störungsquote = (SERVFAIL + Anfragen ohne Antwort) ÷ Anfragen gesamt
# Erfolgsquote  = (NOERROR − NODATA) ÷ (Anfragen gesamt × 0,75)
#                 0,75 = Anteil der Anfragen, die auflösen sollen (50 % intern + 25 % extern)

--- Nützliche Funktionen (englische / deutsche Oberfläche) ------------
# SUM / SUMME                 =SUM(C5:C35)
# AVERAGE / MITTELWERT        =AVERAGE(D5:D12)
# MEDIAN / MEDIAN             =MEDIAN(I5:I35)
# COUNTIF / ZÄHLENWENN        =COUNTIF(E4:E21;"Shop-Anwendung")
# AVERAGEIF / MITTELWERTWENN  =AVERAGEIF(B5:B35;"Mo";C5:C35)
# MAX / MAX                   =MAX(H5:H35)
# Dauer in Stunden aus zwei Zeitpunkten:   =(C4-B4)*24
# Prozent: Zelle markieren -> Format -> Zahlen -> Prozent
# Hinweis: Leere Zellen zählen bei AVERAGE und MEDIAN nicht mit, negative Werte schon.

--- Pareto-Diagramm ---------------------------------------------------
# 1. Kategorien und Anzahl in eine Tabelle (COUNTIF), absteigend sortieren (Daten -> Sortieren)
# 2. Spalte Anteil = Anzahl ÷ Summe, Spalte kumuliert = Summe der Anteile bis zu dieser Zeile
# 3. Kategorie, Anzahl und kumuliert markieren -> Einfügen -> Diagramm -> «Säule und Linie»
# 4. kumuliert als Linie auf die zweite Y-Achse (Datenreihe -> Rechtsklick -> Datenreihe formatieren)
# 5. Titel und Achsenbeschriftungen setzen
