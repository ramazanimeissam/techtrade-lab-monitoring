1. DOCKER-GRUNDLAGEN
----------------------------------------------------------------------
  Ersten Container starten (Port 8090 auf dem Host -> Port 80 im Container):
    Auf SRV-WEB:
      sudo docker images
      sudo docker run -d --name web -p 8090:80 nginx:1.30.5-alpine
      sudo docker ps
      curl -s http://localhost:8090 | head -5

  Container untersuchen:
    Auf SRV-WEB:
      sudo docker logs web
      sudo docker inspect web
      sudo docker inspect --format '{{.State.Status}} {{.State.ExitCode}}' web
      sudo docker inspect --format '{{.NetworkSettings.Ports}}' web
      sudo docker exec web nginx -v
      sudo docker top web

  Lebenszyklus:
    Auf SRV-WEB:
      sudo docker stop web
      sudo docker ps            # leer: zeigt nur laufende Container
      sudo docker ps -a         # web: Exited (0)
      sudo docker start web
      sudo docker rm -f web
      sudo docker ps -a         # web ist gelöscht

  Tags und Digest (zwei Versionen desselben Images):
    Auf SRV-WEB:
      sudo docker images nginx
      sudo docker image inspect --format '{{.RepoDigests}}' nginx:1.30.5-alpine
      sudo docker image inspect --format '{{.RepoDigests}}' nginx:1.30.6-alpine

  Dockerfile anpassen:
    Auf SRV-WEB:
      cd /opt/lab/uebung
      cat Dockerfile
      sudo nano index.html
  In index.html die Zeile <p>Platzhalter</p> ersetzen, z. B.:

>>> Anfang
<p>TechTrade Intranet - Gruppe 1</p>
<<< Ende

    Auf SRV-WEB:
      sudo nano Dockerfile
  Gesamten Inhalt ersetzen durch:

>>> Anfang
# Intranet-Startseite TechTrade AG
# Version fest statt latest: jeder Build liefert dieselbe Basis; slim = kleineres Image
FROM python:3.12-slim
COPY index.html /srv/
WORKDIR /srv
# Dokumentiert den Port der Anwendung (veröffentlicht ihn nicht)

# Nicht als root laufen: bricht die Isolation, hat der Angreifer keine root-Rechte
USER nobody
# Prüft die Funktion (Seite liefert eine Antwort), nicht nur, ob der Prozess lebt
HEALTHCHECK --interval=10s --timeout=3s --retries=3 CMD python -c "import urllib.request; urllib.request.urlopen('http://localhost:8000/')"
CMD ["python", "-m", "http.server", "8000"]
<<< Ende

  Bauen, starten, prüfen:
    Auf SRV-WEB:
      sudo docker build -t uebung:1.0 .
      sudo docker images uebung
      sudo docker run -d --name intranet -p 8000:8000 uebung:1.0
      sudo docker ps            # nach ca. 10 s: (healthy)
      curl -s http://localhost:8000 | grep TechTrade
      sudo docker exec intranet id     # uid=65534(nobody) = nicht root
      sudo docker inspect --format '{{.State.Health.Status}}' intranet

  Aufräumen:
    Auf SRV-WEB:
      sudo docker rm -f intranet

2. SHOP-STACK IN BETRIEB NEHMEN
----------------------------------------------------------------------
  Stack lesen:
    Auf SRV-WEB:
      cd /opt/shop
      cat compose.yaml
      cat .env
      sudo docker compose config      # Datei mit eingesetzten Variablen

  Starten und Status:
    Auf SRV-WEB:
      sudo docker compose up -d
      sudo docker compose ps
      sudo docker compose logs app
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"

  Bestellung von aussen (über Netz, Firewall und Reverse Proxy):
    Firefox auf SRV-MON -> http://10.10.20.13 -> Artikel bestellen
    oder:
    Auf SRV-MON:
      curl -s -o /dev/null -w "%{http_code}\n" -d "artikel=NB-14&menge=1" http://10.10.20.13/bestellen
      # 303 = Bestellung gespeichert, weiter zur Startseite
  Kontrolle: Mailpit zeigt die Entwarnung [RESOLVED] ShopNichtVerfuegbar.

  Netze und Namensauflösung:
    Auf SRV-WEB:
      sudo docker network ls
      sudo docker network inspect shop_backend | grep Name
      sudo docker compose exec app getent hosts db
      sudo docker compose exec proxy ping -c 1 db
      # proxy ist nicht im Netz backend: "bad address 'db'"

  Persistenz:
    Auf SRV-WEB:
      sudo docker compose down          # Container und Netze weg, Volume bleibt
      sudo docker volume ls
      sudo docker compose up -d
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"   # gleicher Stand
    Was down -v bewirkt (löscht das Volume und damit alle Bestellungen):
      sudo docker compose down -v
      sudo docker compose up -d
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"   # wieder 3

3. KONZEPT UMSETZEN: MONITORING UND WARTUNG
----------------------------------------------------------------------
  cAdvisor-Compose vervollständigen:
    Auf SRV-WEB:
      cd /opt/cadvisor
      sudo nano compose.yaml
  Gesamten Inhalt ersetzen durch:

>>> Anfang
# cAdvisor - misst CPU, Arbeitsspeicher, Netz und Disk pro Container (SRV-WEB)
services:
  cadvisor:
    # Version fest statt latest: kein unbemerktes Update beim nächsten Start
    image: ghcr.io/google/cadvisor:0.60.5
    ports:
      # Host-Port 8081 (8080 ist im Betrieb oft belegt), Container-Port bleibt 8080
      - "8081:8080"
    volumes:
      # alle Mounts nur lesend (:ro): cAdvisor liest, verändert nichts auf dem Host
      - /:/rootfs:ro
      - /var/run:/var/run:ro
      - /sys:/sys:ro
      - /var/lib/docker/:/var/lib/docker:ro
      - /dev/disk/:/dev/disk:ro
    devices:
      - /dev/kmsg
    privileged: true
    # startet nach Neustart des Hosts oder Absturz wieder, ausser manuell gestoppt
    restart: unless-stopped
<<< Ende

    Auf SRV-WEB:
      sudo docker compose config -q
      sudo docker compose up -d
      sudo docker compose ps
      curl -s http://localhost:8081/metrics | grep -c container_
      # eine Zahl > 0 = cAdvisor liefert Messwerte

  Prometheus-Job cadvisor:
    Auf SRV-MON:
      sudo nano /etc/prometheus/prometheus.yml
  Unter scrape_configs als weiteren Job einfügen, gleich eingerückt wie
  der Job prometheus (2 Leerzeichen vor "- job_name"):

>>> Anfang
  - job_name: cadvisor
    static_configs:
      - targets: ['10.10.20.13:8081']
<<< Ende

  Zwei Alarmregeln ergänzen:
    Auf SRV-MON:
      sudo nano /etc/prometheus/alerts.yml
  Ganz unten anfügen, gleich eingerückt wie "- alert: ZielNichtErreichbar"
  (6 Leerzeichen vor "- alert"):

>>> Anfang
      - alert: ShopContainerFehlt
        expr: count(container_last_seen{name=~"shop-(proxy|app|db)-1"}) < 3 or absent(container_last_seen{name=~"shop-(proxy|app|db)-1"})
        for: 1m
        labels:
          severity: kritisch
        annotations:
          summary: "Shop-Stack: nicht alle drei Container laufen"
          description: "Seit über 1 Minute fehlt mindestens ein Container (proxy, app, db). Erste Massnahme: docker compose ps auf SRV-WEB."

      - alert: ContainerSpeicherKnapp
        expr: container_memory_working_set_bytes{name=~"shop-.*"} / (container_spec_memory_limit_bytes{name=~"shop-.*"} > 0) * 100 > 80
        for: 1m
        labels:
          severity: warnung
        annotations:
          summary: "{{ $labels.name }} nutzt {{ $value | printf \"%.0f\" }} % seines Speicherlimits"
          description: "Über 80 % des Limits seit über 1 Minute. Erste Massnahme: docker stats, Logs und letzte Änderung prüfen."
<<< Ende

  Prüfen und neu laden:
    Auf SRV-MON:
      sudo docker exec prometheus promtool check rules /config/alerts.yml
      sudo docker exec prometheus promtool check config /config/prometheus.yml
      curl -X POST http://localhost:9090/-/reload
  Ansehen: Prometheus -> Status -> Target health (cadvisor UP)
           Prometheus -> Alerts (neue Regeln Inactive = OK)

  Abfragen in Grafana -> Explore -> prometheus -> Code:
    container_memory_working_set_bytes{name=~"shop-.*"}
      Fragt:    Wie viel Arbeitsspeicher belegt jeder Shop-Container gerade?
      Ergebnis: Bytes pro Container (Einheit bytes(IEC)); app im Leerlauf ca. 20-40 MB.
      Nutzen:   Normalverbrauch kennen, Limit begründen, Speicherleck erkennen.
    rate(container_cpu_usage_seconds_total{name=~"shop-.*"}[1m]) * 100
      Fragt:    Wie viel Prozent einer CPU nutzt jeder Container?
      Ergebnis: Prozent pro Container; im Leerlauf unter 1 %.
      Nutzen:   Verursacher bei hoher Last finden.

  Wartung: Datensicherung vor dem Update:
    Auf SRV-WEB:
      cd /opt/shop
      sudo docker compose exec -T db pg_dump -U shop shop | sudo tee /backup/shop_vor_update.sql > /dev/null
      ls -l /backup
      grep -A 5 "COPY public.bestellungen" /backup/shop_vor_update.sql

  Pre-Check (Ausgangszustand festhalten):
    Auf SRV-WEB:
      sudo docker compose ps
      sudo docker compose exec proxy nginx -v
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"
      curl -sk -o /dev/null -w "%{http_code}\n" https://localhost

  Patch einspielen (nginx 1.30.5 -> 1.30.6):
    Auf SRV-WEB:
      sudo nano compose.yaml
  Beim Dienst proxy die Zeile image ersetzen durch:

>>> Anfang

    image: nginx:1.30.6-alpine
<<< Ende

    Auf SRV-WEB:
      sudo docker compose config -q
      sudo docker compose up -d       # nur proxy wird neu erstellt

  Post-Check (gleiche Prüfungen wie vorher):
    Auf SRV-WEB:

4. STÖRUNGEN BEHEBEN
----------------------------------------------------------------------
  Für jede Störung: Uhrzeit notieren (date), Störung auslösen, auf den
  Alarm in Mailpit warten (MTTD), dann eingrenzen und beheben.

  Evidenz sichern (vor jeder Sofortmassnahme):
    Auf SRV-WEB:
      cd /opt/shop
      sudo docker compose ps -a > /tmp/vorfall_ps.txt
      sudo docker compose logs --no-color --timestamps > /tmp/vorfall_logs.txt
      sudo docker events --since 30m --until 0s > /tmp/vorfall_events.txt

  Eingrenzen von aussen nach innen:
    Auf SRV-WEB:
      curl -s -o /dev/null -w "%{http_code}\n" http://localhost   # 1 Dienst
      sudo docker compose logs --tail 20 proxy                     # 2 Proxy
      sudo docker compose ps -a                                    # 3 Zustand
      sudo docker compose logs --tail 20 app                       # 4 Details
      sudo docker compose exec app getent hosts db                 # 5 Abhängigkeit
      df -h /                                                      # 7 Host
      free -m

  --- Störung: Datenbank gestoppt
    Auslösen:
    Auf SRV-WEB:
      date
      sudo docker compose stop db
    Symptom: Shop zeigt "500 - Interner Fehler".
    Eingrenzen:
      sudo docker compose ps -a                # db: Exited
      sudo docker compose logs --tail 5 app    # "Datenbank db:5432 nicht erreichbar"
    Sofortmassnahme:
      sudo docker compose start db
    Wirkung prüfen:
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"

  --- Störung: Update auf Version 1.5
    Auslösen:
    Auf SRV-WEB:
      date
      sudo nano .env            # SHOP_VERSION=1.5
      sudo docker compose up -d
      for i in $(seq 1 60); do curl -s -o /dev/null http://localhost; done
    Symptom: Shop zeitweise nicht erreichbar (502), dann wieder da.
    Eingrenzen:
      sudo docker compose ps                   # app: Restarting oder "Up xx seconds"
      sudo docker inspect --format 'Neustarts: {{.RestartCount}}' shop-app-1
      sudo docker events --since 10m --until 0s --filter container=shop-app-1
      # "oom" und "die ... exitCode=137" = vom Kernel beendet, Speicherlimit
      sudo docker stats --no-stream
      sudo docker compose logs --tail 10 app   # "Warenkorb-Cache" wächst
    Sofortmassnahme (Rollback auf die letzte bewährte Version):
      sudo nano .env            # SHOP_VERSION=1.4
      sudo docker compose up -d
    Wirkung prüfen:
      sudo docker compose ps
      curl -s http://localhost | grep -o "Version [0-9.]*"
      for i in $(seq 1 60); do curl -s -o /dev/null http://localhost; done
      sudo docker stats --no-stream            # Speicher app bleibt konstant

  --- Störung: Tippfehler in .env
    Auslösen:
    Auf SRV-WEB:
      date
      sudo nano .env            # DB_HOST=dbb
      sudo docker compose restart app
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"   # geht noch!
      sudo docker compose up -d
    Symptom: erst nach "up -d" zeigt der Shop "500 - Interner Fehler".
    Eingrenzen:
      sudo docker compose logs --tail 5 app    # could not translate host name "dbb"
      sudo docker compose config | grep DB_HOST
      sudo docker inspect --format '{{range .Config.Env}}{{println .}}{{end}}' shop-app-1 | grep DB_HOST
    Sofortmassnahme:
      sudo nano .env            # DB_HOST=db
      sudo docker compose up -d
    Wirkung prüfen:
      curl -s http://localhost | grep -o "Bestellungen: [0-9]*"
    Merke: restart startet den bestehenden Container mit alter Konfiguration
    neu; erst up -d erstellt ihn mit der geänderten Konfiguration.

  --- Störung: falscher Port im Reverse Proxy
    Auslösen:
    Auf SRV-WEB:
      date
      sudo nano nginx/default.conf     # proxy_pass http://app:8081;
      sudo docker compose exec proxy nginx -t
      sudo docker compose exec proxy nginx -s reload
    Symptom: Shop zeigt "502 Bad Gateway".
    Eingrenzen:
      curl -s -o /dev/null -w "%{http_code}\n" http://localhost      # 502
      sudo docker compose logs --tail 5 proxy
      # connect() failed (111: Connection refused) ... upstream: "http://...:8081/"
      sudo docker compose exec proxy wget -qO- http://app:8080/health   # ok = App läuft
    Sofortmassnahme:
      sudo nano nginx/default.conf     # proxy_pass http://app:8080;
      sudo docker compose exec proxy nginx -t
      sudo docker compose exec proxy nginx -s reload
    Wirkung prüfen:
      curl -s -o /dev/null -w "%{http_code}\n" http://localhost      # 200

  Ausgangszustand nach allen Störungen:
    Auf SRV-WEB:
      grep -E "SHOP_VERSION|DB_HOST" .env      # 1.4 und db
      grep proxy_pass nginx/default.conf      # app:8080
      sudo docker compose ps                  # drei Container Up
    Prometheus -> Alerts: alles Inactive

5. URSACHENANALYSE (VORFALL VERSION 1.5)
----------------------------------------------------------------------
  Daten für den Zeitstrahl sammeln:
    Auf SRV-WEB:
      cd /opt/shop
      sudo docker events --since 2h --until 0s --filter container=shop-app-1
      sudo docker compose logs --timestamps app | grep -E "startet|Warenkorb" | tail -30
      grep -E "SHOP_VERSION" .env
    Mailpit: Zeitpunkte der Alarm- und Entwarnungsmails
    Alertmanager und Prometheus zeigen Zeiten in UTC (Schweiz: UTC+1, Sommer UTC+2).

  Grafana -> Explore -> prometheus -> Code (Zeitraum: Vorfall):
    container_memory_working_set_bytes{name="shop-app-1"}
      Fragt:    Wie entwickelt sich der Speicher der App über die Zeit?
      Ergebnis: Sägezahn = wächst bis zum Limit (256 MB), fällt beim Neustart auf ca. 30 MB.
      Nutzen:   Beleg für Speicherleck und OOM-Kill.
    changes(container_start_time_seconds{name="shop-app-1"}[15m])
      Fragt:    Wie oft wurde die App in den letzten 15 Minuten neu gestartet?
      Ergebnis: Anzahl Neustarts; im Normalbetrieb 0.
      Nutzen:   Crash-Loop erkennen, auch wenn die Restart-Policy ihn verdeckt.

>>> Anfang
# TechTrade Shop - Container-Stack auf SRV-WEB
# Prüfen:    sudo docker compose config
# Starten:   sudo docker compose up -d
# Status:    sudo docker compose ps
name: shop

services:
  proxy:
    # Wartung: Patch 1.30.5 -> 1.30.6 (Sicherheitskorrekturen), Rückweg: 1.30.5
    image: nginx:1.30.6-alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf:ro
      - /etc/ssl/techtrade:/etc/ssl/techtrade:ro
    depends_on:
      app:
        # Proxy erst starten, wenn die App Anfragen beantwortet
        condition: service_healthy
    # Log-Rotation: höchstens 3 Dateien à 10 MB, damit die Disk nicht vollläuft
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"
    restart: unless-stopped
    networks: [frontend]

  app:
    image: techtrade/shop:${SHOP_VERSION}
    environment:
      DB_HOST: ${DB_HOST}
      DB_PORT: 5432
      DB_USER: shop
      DB_NAME: shop
      DB_PASSWORD_FILE: /run/secrets/db_password
    secrets: [db_password]
    depends_on:
      db:
        # App erst starten, wenn die Datenbank Verbindungen annimmt
        condition: service_healthy
    # Healthcheck: prüft die Funktion (/health fragt die Datenbank ab), nicht nur den Prozess
    healthcheck:
      test: ["CMD", "python", "-c", "import urllib.request; urllib.request.urlopen('http://localhost:8080/health')"]
      interval: 10s
      timeout: 3s
      retries: 3
      start_period: 15s
    deploy:
      resources:
        limits:
          # Speicherlimit: Normalverbrauch ca. 30 MB plus Reserve; schützt den Host vor Speicherlecks
          memory: 256M
    # Log-Rotation: höchstens 3 Dateien à 10 MB
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"
    restart: unless-stopped
    networks: [frontend, backend]

  db:
    image: postgres:16.15-alpine
    environment:
      POSTGRES_USER: shop
      POSTGRES_DB: shop
      POSTGRES_PASSWORD_FILE: /run/secrets/db_password
      TZ: Europe/Zurich
    secrets: [db_password]
    volumes:
      - dbdata:/var/lib/postgresql/data
    # Healthcheck: Datenbank nimmt Verbindungen an
    healthcheck:
      test: ["CMD", "pg_isready", "-U", "shop"]
      interval: 10s
      timeout: 3s
      retries: 5
    # Log-Rotation: höchstens 3 Dateien à 10 MB
    logging:
      driver: json-file
      options:
        max-size: "10m"
        max-file: "3"
    restart: unless-stopped
    networks: [backend]

networks:
  frontend:
  backend:
    internal: true

volumes:
  dbdata:

secrets:
  db_password:
    file: ./secrets/db_password.txt
<<< Ende

  Prüfen und anwenden:

      sudo docker compose ps               # (healthy) bei app und db
